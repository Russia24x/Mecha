   967→export function getArea(areaId: string): AreaData | undefined {
   968→  for (const act of ACTS) {
   969→    for (const region of act.regions) {
   970→      for (const area of region.areas) {
   971→        if (area.id === areaId) return area;
   972→      }
   973→    }
   974→  }
   975→  return undefined;
   976→}
   977→
   978→export function getAllAreas(): AreaData[] {
   979→  const areas: AreaData[] = [];
   980→  for (const act of ACTS) {
   981→    for (const region of act.regions) {
   982→      areas.push(...region.areas);
   983→    }
   984→  }
   985→  return areas;
   986→}
