     1→     1→/**
     2→     2→ * MECHA: LAST PROTOCOL — Area Loader
     3→     3→ * Converts AreaData into Matter physics bodies + visual rectangles + triggers.
     4→     4→ * Reads platform/hazard data from AreaData.sections.
     5→     5→ * Independent of entities — just builds the physical world.
     6→     6→ */
     7→     7→import Phaser from 'phaser';
     8→     8→import { GAME, COLORS } from '../shared/Constants';
     9→     9→import { PhysicsSystem } from '../systems/PhysicsSystem';
    10→    10→import { WorldSystem } from './WorldSystem';

    12→    12→import type { AreaData, SectionData, PlatformData, HazardData, LoreObjectData, LandmarkData } from '../data/types';
    13→    13→import { AreaStrategy, type HazardVisualData, type PlatformType } from './strategies/AreaStrategy';
    14→    14→import { FactoryAreaStrategy } from './strategies/FactoryAreaStrategy';
    15→    15→import { ForestAreaStrategy } from './strategies/ForestAreaStrategy';
    16→    16→import { WastesAreaStrategy } from './strategies/WastesAreaStrategy';
    17→    17→import { CityAreaStrategy } from './strategies/CityAreaStrategy';
    18→    18→
    19→    19→export interface LoadedArea {
    20→    20→  solids: Phaser.Physics.Matter.Image[];
    21→    21→  sectionTriggers: Phaser.Physics.Matter.Image[];
    22→    22→  checkpointTriggers: Phaser.Physics.Matter.Image[];
    23→    23→  bossEntryTrigger: Phaser.Physics.Matter.Image | null;
    24→    24→  hazardTriggers: Phaser.Physics.Matter.Image[];
    25→    25→  visualRects: Phaser.GameObjects.Rectangle[];
    26→    26→  loreObjects: Phaser.GameObjects.Container[];
    27→    27→  landmarks: Phaser.GameObjects.Container[];
    28→    28→  grappleAnchors: Phaser.GameObjects.Container[];
    29→    29→  empDoors: Phaser.GameObjects.Container[];
    30→    30→  shortcuts: Phaser.GameObjects.Container[];
    31→    31→  collectibles: Phaser.GameObjects.Container[];
    32→    32→}
    33→    33→
    34→    34→export class AreaLoader {
    35→    35→  private scene: Phaser.Scene;
    36→    36→  private physics: PhysicsSystem;
    37→    37→  private regionId: string = 'factory';
    38→    38→  private activeTweens: Phaser.Tweens.Tween[] = [];
    39→    39→  private strategy: AreaStrategy | null = null;
    40→    40→
    41→    41→  constructor(scene: Phaser.Scene, physics: PhysicsSystem) {
    42→    42→    this.scene = scene;
    43→    43→    this.physics = physics;
    44→    44→  }
    45→    45→
    46→    46→  /** Create a tween and track it for cleanup on unload. */
    47→    47→  private trackedTween(config: Phaser.Types.Tweens.TweenBuilderConfig): Phaser.Tweens.Tween {
    48→    48→    const t = this.scene.tweens.add(config);
    49→    49→    this.activeTweens.push(t);
    50→    50→    return t;
    51→    51→  }
    52→    52→
    53→    53→  /** Create the appropriate strategy for a region. */
    54→    54→  private createStrategy(regionId: string): AreaStrategy {
    55→    55→    const tweenFn = (config: Phaser.Types.Tweens.TweenBuilderConfig) => this.trackedTween(config);
    56→    56→    if (regionId === 'forest') return new ForestAreaStrategy(this.scene, tweenFn);
    57→    57→    if (regionId === 'wastes') return new WastesAreaStrategy(this.scene, tweenFn);
    58→    58→    if (regionId === 'city') return new CityAreaStrategy(this.scene, tweenFn);
    59→    59→    // Default: factory
    60→    60→    return new FactoryAreaStrategy(this.scene, tweenFn);
    61→    61→  }
    62→    62→
    63→    63→  /**
    64→    64→   * Build the full area: floor, ceiling, per-section platforms, hazards,
    65→    65→   * section triggers, checkpoint triggers, boss entry trigger.
    66→    66→   * RegionId controls ALL visual styling — factory vs forest are completely separate.
    67→    67→   */
    68→    68→  load(area: AreaData): LoadedArea {
    69→    69→    this.regionId = area.regionId;  // ── Store region for all visual decisions
    70→    70→    this.strategy = this.createStrategy(area.regionId);
    71→    71→    const result: LoadedArea = {
    72→    72→      solids: [],
    73→    73→      sectionTriggers: [],
    74→    74→      checkpointTriggers: [],
    75→    75→      bossEntryTrigger: null,
    76→    76→      hazardTriggers: [],
    77→    77→      visualRects: [],
    78→    78→      loreObjects: [],
    79→    79→      landmarks: [],
    80→    80→      grappleAnchors: [],
    81→    81→      empDoors: [],
    82→    82→      shortcuts: [],
    83→    83→      collectibles: [],
    84→    84→    };
    85→    85→
    86→    86→    // Floor (spans entire area)
    87→    87→    this.addSolid(result, area.totalWidth / 2, GAME.HEIGHT, area.totalWidth, 80);
    88→    88→    // Ceiling
    89→    89→    this.addSolid(result, area.totalWidth / 2, -20, area.totalWidth, 40);
    90→    90→
    91→    91→    // Per-section platforms + hazards
    92→    92→    for (const section of area.sections) {
    93→    93→      this.buildSection(result, section, area.sectionWidth);
    94→    94→    }
    95→    95→
    96→    96→    // Section triggers (at start of each section)
    97→    97→    for (let i = 0; i < area.sections.length; i++) {
    98→    98→      const sec = area.sections[i];
    99→    99→      const trigger = this.physics.addSensor(sec.x + 80, GAME.HEIGHT / 2, 40, GAME.HEIGHT, `section-${sec.id}`);
   100→   100→      trigger.setData('sectionId', sec.id);
   101→   101→      result.sectionTriggers.push(trigger);
   102→   102→    }
   103→   103→
   104→   104→    // Checkpoint triggers
   105→   105→    for (const secId of area.checkpointSections) {
   106→   106→      const cpX = (secId - 1) * area.sectionWidth + 640;
   107→   107→      const cp = this.physics.addSensor(cpX, GAME.HEIGHT - 80, 60, 60, `checkpoint-${secId}`);
   108→   108→      cp.setData('isCheckpoint', true);
   109→   109→      cp.setData('checkpointSection', secId);
   110→   110→      result.checkpointTriggers.push(cp);
   111→   111→    }
   112→   112→
   113→   113→    // Boss entry trigger (section 6 = last section with bossId)
   114→   114→    const bossSection = area.sections.find(s => s.bossId);
   115→   115→    if (bossSection) {
   116→   116→      const bossX = bossSection.x + 400;
   117→   117→      const trigger = this.physics.addSensor(bossX, GAME.HEIGHT - 100, 40, 200, 'boss-entry');
   118→   118→      trigger.setData('isBossEntry', true);
   119→   119→      trigger.setData('bossId', bossSection.bossId);
   120→   120→      result.bossEntryTrigger = trigger;
   121→   121→    }
   122→   122→
   123→   123→    return result;
   124→   124→  }
   125→   125→
   126→   126→  private buildSection(result: LoadedArea, section: SectionData, sectionWidth: number): void {
   127→   127→    if (section.platforms) {
   128→   128→      for (const plat of section.platforms) {
   129→   129→        this.addSolid(result, plat.x, plat.y, plat.w, plat.h);
   130→   130→      }
   131→   131→    }
   132→   132→    // Hazards
   133→   133→    if (section.hazards) {
   134→   134→      for (const hazard of section.hazards) {
   135→   135→        const trigger = this.physics.addSensor(hazard.x, hazard.y, hazard.w, hazard.h, `hazard-${hazard.type}`);
   136→   136→        trigger.setData('isHazard', true);
   137→   137→        trigger.setData('hazardDamage', hazard.damage);
   138→   138→        result.hazardTriggers.push(trigger);
   139→   139→        // ── Graphical hazard visuals per type ──
   140→   140→        const hazardVis = this.createHazardVisual(hazard);
   141→   141→        result.visualRects.push(hazardVis as unknown as Phaser.GameObjects.Rectangle);
   142→   142→      }
   143→   143→    }
   144→   144→    // Lore Objects — interactable environmental storytelling
   145→   145→    if (section.loreObjects) {
   146→   146→      for (const lore of section.loreObjects) {
   147→   147→        const container = this.createLoreObject(lore);
   148→   148→        result.loreObjects.push(container);
   149→   149→      }
   150→   150→    }
   151→   151→    // Landmarks — large visual structures
   152→   152→    if (section.landmarks) {
   153→   153→      for (const lm of section.landmarks) {
   154→   154→        const container = this.createLandmark(lm);
   155→   155→        result.landmarks.push(container);
   156→   156→      }
   157→   157→    }
   158→   158→    // Grapple Anchors — ability-gated (Metroidvania)
   159→   159→    if (section.grappleAnchors) {
   160→   160→      for (const anchor of section.grappleAnchors) {
   161→   161→        const container = this.createGrappleAnchor(anchor);
   162→   162→        result.grappleAnchors.push(container);
   163→   163→      }
   164→   164→    }
   165→   165→    // EMP Doors — ability-gated (Metroidvania)
   166→   166→    if (section.empDoors) {
   167→   167→      for (const door of section.empDoors) {
   168→   168→        const container = this.createEmpDoor(door);
   169→   169→        result.empDoors.push(container);
   170→   170→      }
   171→   171→    }
   172→   172→    // Shortcuts — one-way doors (Dark Souls style backtracking)
   173→   173→    if (section.shortcuts) {
   174→   174→      for (const sc of section.shortcuts) {
   175→   175→        const container = this.createShortcut(sc);
   176→   176→        result.shortcuts.push(container);
   177→   177→      }
   178→   178→    }
   179→   179→    // Collectibles — pickups (health/energy fragments, skill points)
   180→   180→    if (section.collectibles) {
   181→   181→      for (const col of section.collectibles) {
   182→   182→        const container = this.createCollectible(col);
   183→   183→        result.collectibles.push(container);
   184→   184→      }
   185→   185→    }
   186→   186→  }

   188→   188→  /** Create a grapple anchor — glowing ring that the player can grapple to. */
   189→   189→  private createGrappleAnchor(anchor: { id: string; x: number; y: number }): Phaser.GameObjects.Container {
   190→   190→    const container = this.scene.add.container(anchor.x, anchor.y);
   191→   191→    // Outer glow ring (pulsing cyan)
   192→   192→    const glow = this.scene.add.circle(0, 0, 14, 0x66f0ff, 0.15);
   193→   193→    glow.setBlendMode(Phaser.BlendModes.ADD);
   194→   194→    container.add(glow);
   195→   195→    this.trackedTween({
   196→   196→      targets: glow, alpha: { from: 0.08, to: 0.25 }, scale: { from: 0.9, to: 1.2 },
   197→   197→      duration: 1200, yoyo: true, repeat: -1, ease: 'Sine.inOut',
   198→   198→    });
   199→   199→    // Inner ring (solid)
   200→   200→    const ring = this.scene.add.circle(0, 0, 8, 0x66f0ff, 0);
   201→   201→    ring.setStrokeStyle(2, 0x66f0ff, 0.8);
   202→   202→    container.add(ring);
   203→   203→    // Center dot
   204→   204→    const dot = this.scene.add.circle(0, 0, 2, 0x66f0ff, 0.9);
   205→   205→    dot.setBlendMode(Phaser.BlendModes.ADD);
   206→   206→    container.add(dot);
   207→   207→    // Rotating spokes (4 small lines)
   208→   208→    for (let i = 0; i < 4; i++) {
   209→   209→      const angle = (i / 4) * Math.PI * 2;
   210→   210→      const spoke = this.scene.add.rectangle(
   211→   211→        Math.cos(angle) * 6, Math.sin(angle) * 6, 4, 1, 0x66f0ff, 0.6
   212→   212→      );
   213→   213→      spoke.setRotation(angle);
   214→   214→      spoke.setBlendMode(Phaser.BlendModes.ADD);
   215→   215→      container.add(spoke);
   216→   216→    }
   217→   217→    // Slow rotation
   218→   218→    this.trackedTween({ targets: container, rotation: Math.PI * 2, duration: 4000, repeat: -1, ease: 'Linear' });
   219→   219→    container.setDepth(6);
   220→   220→    container.setData('grappleAnchorId', anchor.id);
   221→   221→    container.setData('isGrappleAnchor', true);
   222→   222→    return container;
   223→   223→  }

   257→   257→  /** Create a shortcut door — one-way, opens from one side and stays open. */
   258→   258→  private createShortcut(sc: { id: string; x: number; y: number; w: number; h: number; toSection: number; opensFrom: string }): Phaser.GameObjects.Container {
   259→   259→    const container = this.scene.add.container(sc.x, sc.y);
   260→   260→    // Door frame (amber, industrial)
   261→   261→    const frame = this.scene.add.rectangle(0, 0, sc.w + 8, sc.h + 8, 0x1a1814, 0.9);
   262→   262→    frame.setStrokeStyle(2, 0xffc040, 0.6);
   263→   263→    container.add(frame);
   264→   264→    // Door body (closed = solid, open = transparent)
   265→   265→    const body = this.scene.add.rectangle(0, 0, sc.w, sc.h, 0x2a2018, 0.85);
   266→   266→    body.setStrokeStyle(1, 0xffc040, 0.4);
   267→   267→    container.add(body);
   268→   268→    // "Opens from" indicator — small arrow on the correct side
   269→   269→    const arrowColor = 0xffc040;
   270→   270→    let arrowX = 0, arrowY = 0, arrowRot = 0;
   271→   271→    switch (sc.opensFrom) {
   272→   272→      case 'left':  arrowX = -sc.w / 2 - 8; arrowRot = Math.PI; break;
   273→   273→      case 'right': arrowX = sc.w / 2 + 8;  arrowRot = 0; break;
   274→   274→      case 'top':   arrowY = -sc.h / 2 - 8; arrowRot = Math.PI / 2; break;
   275→   275→      case 'bottom': arrowY = sc.h / 2 + 8; arrowRot = -Math.PI / 2; break;
   276→   276→    }
   277→   277→    const arrow = this.scene.add.triangle(arrowX, arrowY, -4, -4, 4, -4, 0, 4, arrowColor, 0.8);
   278→   278→    arrow.setRotation(arrowRot);
   279→   279→    container.add(arrow);
   280→   280→    // Label
   281→   281→    const label = this.scene.add.text(0, 0, '⇌', { fontSize: '12px', color: '#ffc040' }).setOrigin(0.5);
   282→   282→    container.add(label);
   283→   283→
   284→   284→    container.setDepth(6);
   285→   285→    container.setData('shortcutId', sc.id);
   286→   286→    container.setData('isShortcut', true);
   287→   287→    container.setData('shortcutOpen', false);
   288→   288→    container.setData('opensFrom', sc.opensFrom);
   289→   289→    container.setData('toSection', sc.toSection);
   290→   290→    container.setSize(sc.w, sc.h);
   291→   291→    // ── FIX Bug 1: Add physics body so the shortcut door actually blocks the player ──
   292→   292→    const physicsBody = this.physics.addStaticRect(sc.x, sc.y, sc.w, sc.h);
   293→   293→    physicsBody.setData('isShortcutBody', true);
   294→   294→    physicsBody.setData('shortcutId', sc.id);
   295→   295→    container.setData('physicsBody', physicsBody);
   296→   296→    return container;
   297→   297→  }

   367→   367→  private addSolid(result: LoadedArea, x: number, y: number, w: number, h: number): void {
   368→   368→    const s = this.physics.addStaticRect(x, y, w, h);
   369→   369→    const g = this.scene.add.graphics();
   370→   370→    g.setDepth(5);
   371→   371→
   372→   372→    // ── Determine platform type ──
   373→   373→    const isWall = h > 80 && w < 100;
   374→   374→    const isFloor = h <= 30 && w >= 100;
   375→   375→    const isLedge = h <= 30 && w < 100;
   376→   376→    const isPillar = h > 80 && w >= 100;
   377→   377→
   378→   378→    let platformType: PlatformType = 'generic';
   379→   379→    if (isWall) platformType = 'wall';
   380→   380→    else if (isFloor) platformType = 'floor';
   381→   381→    else if (isLedge) platformType = 'ledge';
   382→   382→    else if (isPillar) platformType = 'pillar';
   383→   383→
   384→   384→    // ── Delegate platform drawing to strategy ──
   385→   385→    this.strategy?.drawPlatform(g, w, h, platformType);
   386→   386→
   387→   387→    g.setPosition(x, y);
   388→   388→    // ⚠️ Store width/height as custom data so VisualCuller can use
   389→   389→    // bounding-box culling. Phaser 4 Graphics doesn't have setSize()
   390→   390→    // (only ComputedSize, which Graphics doesn't include), so we store
   391→   391→    // the platform dimensions on the data manager instead.
   392→   392→    // Without this, culler uses center-point only — wide platforms
   393→   393→    // (e.g. 600px floor) would be culled when their center is off-screen
   394→   394→    // even though part of them is still visible.

   408→   408→  private createHazardVisual(hazard: { type: string; x: number; y: number; w: number; h: number; damage: number }): Phaser.GameObjects.Container {

   636→   636→  /** Destroy all loaded area objects. */
   637→   637→  unload(loaded: LoadedArea): void {
   638→   638→    // ── Stop all tracked tweens (sparks, fire, steam, lore pulses, etc.) ──
   639→   639→    this.activeTweens.forEach(t => { if (t && t.isPlaying()) t.stop(); });
   640→   640→    this.activeTweens = [];
   641→   641→    // Clean up hazard timers (electrical sparks, fire smoke, steam) before destroying visuals
   642→   642→    loaded.visualRects.forEach(v => {
   643→   643→      if (!v) return;
   644→   644→      const obj = v as unknown as { __sparkTimer?: Phaser.Time.TimerEvent; __smokeTimer?: Phaser.Time.TimerEvent; __steamTimer?: Phaser.Time.TimerEvent };
   645→   645→      obj.__sparkTimer?.remove();
   646→   646→      obj.__smokeTimer?.remove();
   647→   647→      obj.__steamTimer?.remove();
   648→   648→      if (v.active) v.destroy();
   649→   649→    });
   650→   650→    loaded.solids.forEach(s => { if (s && s.active) s.destroy(); });
   651→   651→    loaded.sectionTriggers.forEach(t => { if (t && t.active) t.destroy(); });
   652→   652→    loaded.checkpointTriggers.forEach(t => { if (t && t.active) t.destroy(); });
   653→   653→    loaded.hazardTriggers.forEach(t => { if (t && t.active) t.destroy(); });
   654→   654→    loaded.bossEntryTrigger?.destroy();
   655→   655→    loaded.loreObjects.forEach(l => { if (l && l.active) l.destroy(); });
   656→   656→    loaded.landmarks.forEach(l => { if (l && l.active) l.destroy(); });
   657→   657→    loaded.grappleAnchors.forEach(a => { if (a && a.active) a.destroy(); });
   658→   658→    // ── FIX Bug 1: Clean up physics bodies from EMP doors + shortcuts (not auto-destroyed) ──
   659→   659→    loaded.empDoors.forEach(d => {
   660→   660→      if (!d) return;
   661→   661→      const pb = d.getData('physicsBody') as Phaser.Physics.Matter.Image | null;
   662→   662→      if (pb && pb.active) {
   663→   663→        try { this.scene.matter.world.remove(pb.body as MatterJS.Body); } catch { /* */ }
   664→   664→        pb.destroy();
   665→   665→      }
   666→   666→      if (d.active) d.destroy();
   667→   667→    });
   668→   668→    loaded.shortcuts.forEach(s => {
   669→   669→      if (!s) return;
   670→   670→      const pb = s.getData('physicsBody') as Phaser.Physics.Matter.Image | null;
   671→   671→      if (pb && pb.active) {
   672→   672→        try { this.scene.matter.world.remove(pb.body as MatterJS.Body); } catch { /* */ }
   673→   673→        pb.destroy();
   674→   674→      }
   675→   675→      if (s.active) s.destroy();
   676→   676→    });
   677→   677→    loaded.collectibles.forEach(c => { if (c && c.active) c.destroy(); });
   678→   678→    loaded.solids = [];
   679→   679→    loaded.sectionTriggers = [];
   680→   680→    loaded.checkpointTriggers = [];
   681→   681→    loaded.hazardTriggers = [];
   682→   682→    loaded.bossEntryTrigger = null;
   683→   683→    loaded.visualRects = [];
   684→   684→    loaded.loreObjects = [];
   685→   685→    loaded.landmarks = [];
   686→   686→    loaded.grappleAnchors = [];
   687→   687→    loaded.empDoors = [];
   688→   688→    loaded.shortcuts = [];
   689→   689→    loaded.collectibles = [];
   690→   690→  }
