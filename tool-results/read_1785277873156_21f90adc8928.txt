   918→export function getArea(areaId: string): AreaData | undefined {
   919→  for (const act of ACTS) {
   920→    for (const region of act.regions) {
   921→      for (const area of region.areas) {
   922→        if (area.id === areaId) return area;
   923→      }
   924→    }
   925→  }
   926→  return undefined;
   927→}

   929→export function getAllAreas(): AreaData[] {
   930→  const areas: AreaData[] = [];
   931→  for (const act of ACTS) {
   932→    for (const region of act.regions) {
   933→      areas.push(...region.areas);
   934→    }
   935→  }
   936→  return areas;
   937→}
