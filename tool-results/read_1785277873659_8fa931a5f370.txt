     1→/**
     2→ * MECHA: LAST PROTOCOL — ParallaxBackground v1.0
     3→ *
     4→ * Theme-specific multi-layer parallax backgrounds that give each region a
     5→ * clear sense of PLACE. NO MORE EMPTY METAL PLATFORMS IN A BLACK VOID.
     6→ *
     7→ * Themes:
     8→ *   - factory: Industrial decay — distant smokestacks, hanging cables,
     9→ *     broken pipes, flickering monitor banks, distant mech silhouettes.
    10→ *   - forest:  Nature reclaiming — distant dead trees, hanging vines,
    11→ *     glowing spores, broken statues overgrown with moss.
    12→ *
    13→ * Architecture (per Phaser 4 sprites-and-images + cameras skill):
    14→ *   - Multiple tileable layers with different scrollFactor values
    15→ *   - Graphics objects procedurally drawn (no external assets)
    16→ *   - Front layer (closest to camera) at depth 1
    17→ *   - Mid layer at depth 0
    18→ *   - Far layer at depth -1
    19→ *   - Sky/background at depth -2
    20→ *
    21→ * Lifecycle:
    22→ *   - build() creates all layers for the area
    23→ *   - destroy() removes them all (called in cleanupPlay)
    24→ *   - Tied to PLAY state only — never appears in hub/menu (effect separation)
    25→ */
    26→import Phaser from 'phaser';
    27→import { GAME } from '../../shared/Constants';
    28→
    29→export type RegionTheme = 'factory' | 'forest' | 'wastes' | 'city' | 'generic';
    30→
    31→interface LayerConfig {
    32→  scrollX: number;   // parallax factor (0 = static, 1 = full camera follow)
    33→  scrollY: number;
    34→  depth: number;
    35→  alpha: number;
    36→}
    37→
    38→export class ParallaxBackground {
    39→  private scene: Phaser.Scene;
    40→  private layers: Phaser.GameObjects.GameObject[] = [];
    41→  private theme: RegionTheme;
    42→  private worldWidth: number;
    43→  private tweens: Phaser.Tweens.Tween[] = [];
    44→  /** For city theme: which pair of bg images to use (0=bg_1/2, 2=bg_3/4) */
    45→  bgStartIndex = 0;
    46→
    47→  constructor(scene: Phaser.Scene, theme: RegionTheme, worldWidth: number) {
    48→    this.scene = scene;
    49→    this.theme = theme;
    50→    this.worldWidth = worldWidth;
    51→  }
    52→
    53→  /** Build all parallax layers. Call once when entering play state. */
    54→  build(): void {
    55→    // === SKY (depth -2, fully static — base color wash) ===
    56→    this.buildSky();
    57→
    58→    // === BACKGROUND ART (user-provided images, tiled across world) ===
    59→    this.buildBackgroundArt();
    60→
    61→    // ⚠️ Stage 2.2: Wastes + City skip procedural Far/Mid/Near silhouette layers.
    62→    // The painted backdrop art at depth -1.5 already contains
    63→    // all depth/silhouettes the artist intended. The generic Far/Mid/Near layers
    64→    // (dark rectangles at depths -1, 0, 1) render ON TOP of the painted art,
    65→    // creating visible flat dark bands that hide the backdrop.
    66→    // Factory and Forest keep these layers (no painted backdrops yet).
    67→    if (this.theme === 'wastes' || this.theme === 'city') return;
    68→
    69→    // === FAR layer (depth -1, scrollFactor 0.1) ===
    70→    const farCfg: LayerConfig = { scrollX: 0.1, scrollY: 0.05, depth: -1, alpha: 0.5 };
    71→    if (this.theme === 'factory') this.buildFactoryFar(farCfg);
    72→    else if (this.theme === 'forest') this.buildForestFar(farCfg);
    73→    else this.buildGenericFar(farCfg);
    74→
    75→    // === MID layer (depth 0, scrollFactor 0.4) ===
    76→    const midCfg: LayerConfig = { scrollX: 0.4, scrollY: 0.2, depth: 0, alpha: 0.7 };
    77→    if (this.theme === 'factory') this.buildFactoryMid(midCfg);
    78→    else if (this.theme === 'forest') this.buildForestMid(midCfg);
    79→    else this.buildGenericMid(midCfg);
    80→
    81→    // === NEAR layer (depth 1, scrollFactor 0.7) — in front of platforms, behind player ===
    82→    const nearCfg: LayerConfig = { scrollX: 0.7, scrollY: 0.3, depth: 1, alpha: 0.85 };
    83→    if (this.theme === 'factory') this.buildFactoryNear(nearCfg);
    84→    else if (this.theme === 'forest') this.buildForestNear(nearCfg);
    85→    else this.buildGenericNear(nearCfg);
    86→  }
    87→
    88→  // ─── SKY ────────────────────────────────────────────────────────────────
    89→  private buildSky(): void {
    90→    // ⚠️ Stage 2.2: Wastes + City sky tint disabled — painted backdrop art provides
    91→    // full-screen atmospheric color. Procedural sky gradient was over-darkening
    92→    // the painted art and shifting its hue away from artist intent.
    93→    // Factory and Forest keep their sky (no painted backdrops yet).
    94→    if (this.theme === 'wastes' || this.theme === 'city') return;
    95→
    96→    // Generate sky as a texture ONCE (not 720 fillRect calls per render)
    97→    const w = GAME.WIDTH, h = GAME.HEIGHT;
    98→    const g = this.scene.add.graphics();
    99→
   100→    if (this.theme === 'factory') {
   101→      // Smoggy amber-brown gradient
   102→      for (let y = 0; y < h; y += 2) {  // step by 2 = half the draw calls
   103→        const t = y / h;
   104→        const r = Math.floor(8 + t * 20);
   105→        const gg = Math.floor(7 + t * 10);
   106→        const b = Math.floor(13 - t * 5);
   107→        g.fillStyle((r << 16) | (gg << 8) | b, 1);
   108→        g.fillRect(0, y, w, 2);
   109→      }
   110→      g.fillStyle(0xffc040, 0.04);
   111→      g.fillCircle(w * 0.7, h * 0.75, 200);
   112→      g.fillStyle(0xff6020, 0.03);
   113→      g.fillCircle(w * 0.3, h * 0.7, 180);
   114→    } else if (this.theme === 'forest') {
   115→      for (let y = 0; y < h; y += 2) {
   116→        const t = y / h;
   117→        const r = Math.floor(5 + t * 8);
   118→        const gg = Math.floor(15 + t * 20);
   119→        const b = Math.floor(10 + t * 5);
   120→        g.fillStyle((r << 16) | (gg << 8) | b, 1);
   121→        g.fillRect(0, y, w, 2);
   122→      }
   123→      g.fillStyle(0x40ff80, 0.025);
   124→      g.fillCircle(w * 0.5, h * 0.4, 250);
   125→    } else {
   126→      // Wastes — sickly green-gray gradient (re-enabled per user request)
   127→      for (let y = 0; y < h; y += 2) {
   128→        const t = y / h;
   129→        const r = Math.floor(8 + t * 10);
   130→        const gg = Math.floor(12 + t * 8);
   131→        const b = Math.floor(6 + t * 4);
   132→        g.fillStyle((r << 16) | (gg << 8) | b, 1);
   133→        g.fillRect(0, y, w, 2);
   134→      }
   135→      g.fillStyle(0x4a5a40, 0.03);
   136→      g.fillCircle(w * 0.5, h * 0.5, 300);
   137→    }
   138→
   139→    // Generate texture from graphics — rendered once, reused as image
   140→    g.generateTexture('__sky_' + this.theme, w, h);
   141→    g.destroy();
   142→
   143→    // Create image from texture — much cheaper to render than Graphics
   144→    const sky = this.scene.add.image(0, 0, '__sky_' + this.theme);
   145→    sky.setOrigin(0, 0);
   146→    sky.setDepth(-1.4);  // above background (-1.5), below far layer (-1)
   147→    sky.setScrollFactor(0);
   148→    sky.setAlpha(0.3);   // semi-transparent tint over background
   149→    this.layers.push(sky);
   150→  }
   151→
   152→  /**
   153→   * Build background art layer using user-provided images.
   154→   * The images are tiled horizontally across the entire world width
   155→   * with a slow parallax scroll factor. This gives the world a real
   156→   * painted/artistic backdrop instead of just procedural graphics.
   157→   */
   158→  private buildBackgroundArt(): void {
   159→    // Factory, Wastes, and City have painted background art
   160→    if (this.theme !== 'factory' && this.theme !== 'wastes' && this.theme !== 'city') return;
   161→
   162→    // Determine texture keys based on theme + bgStartIndex (per-area selection).
   163→    // Per user request (round-19):
   164→    //   Act I (factory): area 1 → factory_bg_1, area 2 → factory_bg_2, area 3 (boss) → factory_bg_3
   165→    //   Act II (wastes): area 1 → wastes_bg_1, area 2 → wastes_bg_2, area 3 (boss) → wastes_bg_3
   166→    //   Act III (city): area 1 → city_bg_1, area 2 → city_bg_2 (double width), area 3 → city_bg_3+city_bg_4 (boss)
   167→    const bgKeys: string[] = [];
   168→    if (this.theme === 'factory') {
   169→      // bgStartIndex: 0=factory_1, 1=factory_2, 2=factory_3(boss)
   170→      const key = `factory_bg_${this.bgStartIndex + 1}`;
   171→      if (!this.scene.textures.exists(key)) return;
   172→      bgKeys.push(key);
   173→    } else if (this.theme === 'wastes') {
   174→      // bgStartIndex: 0=wastes_1, 1=wastes_2, 2=wastes_3(boss)
   175→      const key = `wastes_bg_${this.bgStartIndex + 1}`;
   176→      if (!this.scene.textures.exists(key)) return;
   177→      bgKeys.push(key);
   178→    } else if (this.theme === 'city') {
   179→      // Per user request (round-20):
   180→      //   ward_1 → city_bg_1
   181→      //   ward_2 → city_bg_2 + city_bg_3 (double width area, two images tiled)
   182→      //   courthouse (boss) → city_bg_4 only
   183→      if (this.bgStartIndex === 0) {
   184→        if (!this.scene.textures.exists('city_bg_1')) return;
   185→        bgKeys.push('city_bg_1');
   186→      } else if (this.bgStartIndex === 1) {
   187→        // ward_2: two images for double-width area
   188→        if (!this.scene.textures.exists('city_bg_2')) return;
   189→        bgKeys.push('city_bg_2');
   190→        if (this.scene.textures.exists('city_bg_3')) bgKeys.push('city_bg_3');
   191→      } else if (this.bgStartIndex === 2) {
   192→        // courthouse (boss): only city_bg_4
   193→        if (!this.scene.textures.exists('city_bg_4')) return;
   194→        bgKeys.push('city_bg_4');
   195→      }
   196→    }
   197→    if (bgKeys.length === 0) return;
   198→
   199→    const textureKey = bgKeys[0];
   200→    const tex = this.scene.textures.get(textureKey);
   201→    const imgW = tex.getSourceImage().width;
   202→    const imgH = tex.getSourceImage().height;
   203→    const targetH = GAME.HEIGHT;
   204→    const scale = targetH / imgH;  // scale to fit screen height
   205→
   206→    // ── City theme: use standard tiling (same as factory/wastes) ──
   207→    // Each Act III area has 2 images that tile across its width.
   208→    // This is the proven Act I pattern — no fixed placement, just cycle.
   209→    const tileW = imgW * scale;
   210→
   211→    // Tile across the world width
   212→    const tileCount = Math.ceil(this.worldWidth / tileW) + 1;
   213→    const container = this.scene.add.container(0, 0);
   214→    container.setDepth(-1.5);  // between sky (-2) and far layer (-1)
   215→    // ⚠️ Per Phaser 4 groups-and-containers skill: pass `true` as the 3rd arg
   216→    // so scrollFactor propagates to all child Images/Graphics. Without it,
   217→    // children keep scrollFactor=1 (full camera follow) and only the container
   218→    // itself parallaxes — children appear to "swim" relative to the container
   219→    // and parallax is broken in practice.
   220→    container.setScrollFactor(0.15, 0.05, true);  // slow parallax, propagate to children
   221→    container.setAlpha(this.theme === 'wastes' ? 0.7 : this.theme === 'city' ? 0.7 : 0.65);  // city/wastes slightly more visible
   222→
   223→    for (let i = 0; i < tileCount; i++) {
   224→      const x = i * tileW;
   225→      // For city theme: use FIXED image order (not cycling) so each image
   226→      // covers a specific part of the act:
   227→      //   Image 1 (bg_1) → beginning (sections 1-2) — industrial shipyard
   228→      //   Image 2 (bg_2) → middle (sections 3-4) — dense metropolis
   229→      //   Image 3 (bg_3) → upper-middle (sections 5-6) — sprawling skyline
   230→      //   Image 4 (bg_4) → end (sections 7-8) — boss arena (neon slums)
   231→      // For factory/wastes: cycle as before.
   232→      let key: string;
   233→      if (this.theme === 'city' && bgKeys.length === 4) {
   234→        // Map tile index to image based on position in world width
   235→        const worldPct = x / this.worldWidth;
   236→        if (worldPct < 0.30) key = bgKeys[0];       // sections 1-2
   237→        else if (worldPct < 0.55) key = bgKeys[1];  // sections 3-4
   238→        else if (worldPct < 0.78) key = bgKeys[2];  // sections 5-6
   239→        else key = bgKeys[3];                         // sections 7-8 (boss)
   240→      } else {
   241→        key = bgKeys[i % bgKeys.length];
   242→      }
   243→      const img = this.scene.add.image(x, GAME.HEIGHT / 2, key);
   244→      img.setOrigin(0, 0.5);
   245→      img.setScale(scale);
   246→      // Flip every other tile for seamless tiling (not for city bg_4 — different aspect)
   247→      if (i % 2 === 1 && this.theme !== 'city') {
   248→        img.setFlipX(true);
   249→      }
   250→      container.add(img);
   251→
   252→      // ── Cover seams between tiles with a dark gradient strip ──
   253→      // Skip seam covers for city — painted art is continuous and seams
   254→      // would create visible dark bands that break the artist's vision.
   255→      if (i > 0 && this.theme !== 'city') {
   256→        const seamX = x;
   257→        // Dark gradient strip at seam (blends left and right tiles)
   258→        const seam = this.scene.add.graphics();
   259→        seam.setDepth(-1.4);  // slightly above background
   260→        seam.fillStyle(0x000000, 0.3);
   261→        seam.fillRect(seamX - 30, 0, 60, GAME.HEIGHT);
   262→        // Gradient fade
   263→        for (let g = 0; g < 6; g++) {
   264→          seam.fillStyle(0x000000, 0.15 - g * 0.02);
   265→          seam.fillRect(seamX - 30 + g * 10, 0, 10, GAME.HEIGHT);
   266→          seam.fillRect(seamX + 30 - g * 10, 0, 10, GAME.HEIGHT);
   267→        }
   268→        container.add(seam);
   269→
   270→        // Fog wisp at seam (re-enabled per user request — adds atmospheric cover)
   271→        // Limit to every 3rd seam to avoid excessive tweens on large worlds
   272→        if (this.theme === 'wastes' && i % 3 === 0) {
   273→          const fogSeam = this.scene.add.circle(seamX, GAME.HEIGHT * 0.4, 60, 0x5a6a50, 0.08);
   274→          fogSeam.setBlendMode(Phaser.BlendModes.ADD);
   275→          fogSeam.setDepth(-1.3);
   276→          container.add(fogSeam);
   277→          this.tweens.push(this.scene.tweens.add({
   278→            targets: fogSeam,
   279→            alpha: { from: 0.04, to: 0.12 },
   280→            scale: { from: 0.8, to: 1.3 },
   281→            y: { from: GAME.HEIGHT * 0.35, to: GAME.HEIGHT * 0.45 },
   282→            duration: 4000 + Math.random() * 2000,
   283→            yoyo: true, repeat: -1, ease: 'Sine.inOut',
   284→          }));
   285→        }
   286→      }
   287→    }
   288→    this.layers.push(container);
   289→
   290→    // Subtle drift tween
   291→    this.tweens.push(this.scene.tweens.add({
   292→      targets: container, alpha: { from: 0.55, to: 0.75 },
   293→      duration: 5000, yoyo: true, repeat: -1, ease: 'Sine.inOut',
   294→    }));
   295→  }

   297→  // ─── FACTORY: FAR — distant smokestacks + skyline ───────────────────────
   298→  private buildFactoryFar(cfg: LayerConfig): void {
   299→    const g = this.scene.add.graphics();
   300→    g.setDepth(cfg.depth); g.setAlpha(cfg.alpha);
   301→    g.setScrollFactor(cfg.scrollX, cfg.scrollY);
   302→
   303→    // Repeat skyline across the whole world width
   304→    const tileW = 600;
   305→    const tiles = Math.ceil(this.worldWidth / tileW) + 1;
   306→    for (let t = 0; t < tiles; t++) {
   307→      const baseX = t * tileW;
   308→      // Distant smokestacks (3 per tile, varied heights)
   309→      for (let i = 0; i < 4; i++) {
   310→        const x = baseX + 80 + i * 130 + Math.sin(t * 7 + i * 3) * 30;
   311→        const stackH = 200 + Math.sin(t * 11 + i * 5) * 80;
   312→        const stackW = 30 + (i % 2) * 10;
   313→        const yTop = GAME.HEIGHT - stackH;
   314→        // Stack body
   315→        g.fillStyle(0x1a1410, 1); g.fillRect(x, yTop, stackW, stackH);
   316→        g.fillStyle(0x2a2018, 1); g.fillRect(x, yTop, 4, stackH);
   317→        // Top cap
   318→        g.fillStyle(0x0a0805, 1); g.fillRect(x - 4, yTop - 6, stackW + 8, 6);
   319→        // Smoke plume (faint, drifting)
   320→        g.fillStyle(0x2a2218, 0.15);
   321→        for (let s = 0; s < 3; s++) {
   322→          g.fillCircle(x + stackW / 2 + Math.sin(s) * 8, yTop - 10 - s * 14, 10 + s * 4);
   323→        }
   324→      }
   325→      // Distant broken factory wall silhouette
   326→      g.fillStyle(0x0a0805, 1);
   327→      g.fillRect(baseX, GAME.HEIGHT - 280, tileW, 280);
   328→      // Window gaps (faint amber — long-dead lights)
   329→      for (let wy = 0; wy < 4; wy++) {
   330→        for (let wx = 0; wx < 6; wx++) {
   331→          if (Math.random() < 0.25) {
   332→            g.fillStyle(0xff8030, 0.15 + Math.random() * 0.1);
   333→            g.fillRect(baseX + 60 + wx * 90, GAME.HEIGHT - 250 + wy * 50, 30, 20);
   334→          }
   335→        }
   336→      }
   337→    }
   338→    this.layers.push(g);
   339→
   340→    // Animate smoke (slow drift) — tween the whole layer's alpha slightly
   341→    this.tweens.push(this.scene.tweens.add({
   342→      targets: g, alpha: { from: cfg.alpha * 0.9, to: cfg.alpha * 1.05 },
   343→      duration: 4000, yoyo: true, repeat: -1, ease: 'Sine.inOut',
   344→    }));
   345→  }
   346→
   347→  // ─── FACTORY: MID — hanging cables, broken pipes, hanging mechs ────────
   348→  private buildFactoryMid(cfg: LayerConfig): void {
   349→    const container = this.scene.add.container(0, 0);
   350→    container.setDepth(cfg.depth); container.setAlpha(cfg.alpha);
   351→    // ⚠️ Propagate scrollFactor to children (cables, pipes, drips, mechs, eyes).
   352→    // Without `true`, each child keeps scrollFactor=1 — children appear to
   353→    // "swim" relative to the container and parallax is broken in practice.
   354→    container.setScrollFactor(cfg.scrollX, cfg.scrollY, true);
   355→
   356→    // Hanging cables (from ceiling)
   357→    const cableCount = Math.ceil(this.worldWidth / 350);
   358→    for (let i = 0; i < cableCount; i++) {
   359→      const x = i * 350 + Math.random() * 80;
   360→      const len = 80 + Math.random() * 120;
   361→      const cable = this.scene.add.graphics();
   362→      cable.lineStyle(2, 0x1a1814, 0.7);
   363→      cable.beginPath();
   364→      let cy = 0;
   365→      cable.moveTo(x, cy);
   366→      for (let s = 0; s < 5; s++) {
   367→        cy += len / 5;
   368→        const sway = Math.sin(s) * 4;
   369→        cable.lineTo(x + sway, cy);
   370→      }
   371→      cable.strokePath();
   372→      // End fitting
   373→      cable.fillStyle(0x2a2820, 0.8);
   374→      cable.fillCircle(x, len, 4);
   375→      container.add(cable);
   376→      // Gentle sway
   377→      this.tweens.push(this.scene.tweens.add({
   378→        targets: cable, x: x + 3, duration: 3000 + Math.random() * 2000, yoyo: true, repeat: -1, ease: 'Sine.inOut',
   379→      }));
   380→    }
   381→
   382→    // Broken pipes (horizontal, mid-frame)
   383→    const pipeCount = Math.ceil(this.worldWidth / 500);
   384→    for (let i = 0; i < pipeCount; i++) {
   385→      const x = i * 500 + 100;
   386→      const y = 100 + Math.random() * 100;
   387→      const w = 100 + Math.random() * 80;
   388→      const pipe = this.scene.add.graphics();
   389→      pipe.fillStyle(0x2a2820, 0.8); pipe.fillRect(x, y, w, 12);
   390→      pipe.fillStyle(0x1a1814, 0.9); pipe.fillRect(x, y + 10, w, 2);
   391→      pipe.fillStyle(0x3a3830, 0.6); pipe.fillRect(x, y, w, 1);
   392→      // Break point (gap + dripping)
   393→      pipe.fillStyle(0x0a0805, 1); pipe.fillRect(x + w * 0.7, y, 20, 12);
   394→      // Drip
   395→      const drip = this.scene.add.circle(x + w * 0.75, y + 14, 2, 0x4a3a2a, 0.6);
   396→      drip.setBlendMode(Phaser.BlendModes.ADD);
   397→      container.add(pipe); container.add(drip);
   398→      this.tweens.push(this.scene.tweens.add({
   399→        targets: drip, y: y + 60, alpha: 0, duration: 2000, repeat: -1, delay: Math.random() * 2000,
   400→        onComplete: (_t, targets) => { (targets[0] as Phaser.GameObjects.Arc).setY(y + 14).setAlpha(0.6); },
   401→      }));
   402→    }
   403→
   404→    // Hanging broken mech silhouettes (faded, swinging gently)
   405→    const mechCount = Math.ceil(this.worldWidth / 800);
   406→    for (let i = 0; i < mechCount; i++) {
   407→      const x = i * 800 + 200 + Math.random() * 100;
   408→      const y = 120 + Math.random() * 60;
   409→      const silhouette = this.scene.add.graphics();
   410→      silhouette.fillStyle(0x1a1814, 0.6);
   411→      // Body
   412→      silhouette.fillRect(x - 12, y, 24, 30);
   413→      // Head
   414→      silhouette.fillRect(x - 8, y - 8, 16, 10);
   415→      // One arm hanging
   416→      silhouette.fillRect(x - 18, y + 4, 6, 20);
   417→      // Cable from ceiling
   418→      silhouette.lineStyle(1, 0x2a2820, 0.5);
   419→      silhouette.beginPath(); silhouette.moveTo(x, 0); silhouette.lineTo(x, y); silhouette.strokePath();
   420→      // Dead eye (red dot, flickering)
   421→      const eye = this.scene.add.circle(x - 2, y - 4, 1, 0xff3030, 0.5);
   422→      eye.setBlendMode(Phaser.BlendModes.ADD);
   423→      container.add(silhouette); container.add(eye);
   424→      this.tweens.push(this.scene.tweens.add({
   425→        targets: silhouette, rotation: 0.04, duration: 4000 + i * 500, yoyo: true, repeat: -1, ease: 'Sine.inOut',
   426→      }));
   427→      this.tweens.push(this.scene.tweens.add({
   428→        targets: eye, alpha: { from: 0.1, to: 0.5 }, duration: 800 + Math.random() * 600, yoyo: true, repeat: -1,
   429→      }));

   484→  // ─── FOREST: FAR — distant dead trees + ancient ruins ───────────────────
   485→  private buildForestFar(cfg: LayerConfig): void {
   486→    const g = this.scene.add.graphics();
   487→    g.setDepth(cfg.depth); g.setAlpha(cfg.alpha);
   488→    g.setScrollFactor(cfg.scrollX, cfg.scrollY);
   489→
   490→    const tileW = 700;
   491→    const tiles = Math.ceil(this.worldWidth / tileW) + 1;
   492→    for (let t = 0; t < tiles; t++) {
   493→      const baseX = t * tileW;
   494→      // Distant dead trees (tall thin silhouettes)
   495→      for (let i = 0; i < 6; i++) {
   496→        const x = baseX + i * 110 + Math.sin(t * 9 + i * 4) * 40;
   497→        const treeH = 280 + Math.sin(t * 13 + i * 7) * 60;
   498→        const trunkW = 14 + (i % 2) * 4;
   499→        const yTop = GAME.HEIGHT - treeH;
   500→        // Trunk
   501→        g.fillStyle(0x0a1410, 1); g.fillRect(x, yTop, trunkW, treeH);
   502→        g.fillStyle(0x142018, 0.8); g.fillRect(x, yTop, 3, treeH);
   503→        // Bare branches (a few)
   504→        g.lineStyle(3, 0x0a1410, 0.7);
   505→        for (let b = 0; b < 4; b++) {
   506→          const by = yTop + 30 + b * 40;
   507→          const dir = b % 2 === 0 ? -1 : 1;
   508→          g.beginPath(); g.moveTo(x + trunkW / 2, by);
   509→          g.lineTo(x + trunkW / 2 + dir * (20 + b * 5), by - 15); g.strokePath();
   510→        }
   511→      }
   512→      // Distant ruined archway (moss-covered)
   513→      if (t % 2 === 0) {
   514→        g.fillStyle(0x1a2818, 0.7);
   515→        const ax = baseX + 300;
   516→        const ay = GAME.HEIGHT - 200;
   517→        g.fillRect(ax, ay, 14, 200);
   518→        g.fillRect(ax + 80, ay, 14, 200);
   519→        g.fillRect(ax, ay, 94, 16);
   520→        // Moss
   521→        g.fillStyle(0x2a4a30, 0.4);
   522→        g.fillRect(ax, ay + 50, 14, 80);
   523→        g.fillRect(ax + 80, ay + 30, 14, 60);
   524→      }
   525→    }
   526→    this.layers.push(g);
   527→  }

   529→  // ─── FOREST: MID — hanging vines, glowing spores ───────────────────────
   530→  private buildForestMid(cfg: LayerConfig): void {
   531→    const container = this.scene.add.container(0, 0);
   532→    container.setDepth(cfg.depth); container.setAlpha(cfg.alpha);
   533→    // ⚠️ Propagate scrollFactor to children (vines, spores).
   534→    container.setScrollFactor(cfg.scrollX, cfg.scrollY, true);
   535→
   536→    // Hanging vines (from ceiling)
   537→    const vineCount = Math.ceil(this.worldWidth / 250);
   538→    for (let i = 0; i < vineCount; i++) {
   539→      const x = i * 250 + Math.random() * 80;
   540→      const len = 100 + Math.random() * 80;
   541→      const vine = this.scene.add.graphics();
   542→      vine.lineStyle(3, 0x2a4a20, 0.7);
   543→      let vy = 0;
   544→      vine.beginPath(); vine.moveTo(x, vy);
   545→      for (let s = 0; s < 6; s++) {
   546→        vy += len / 6;
   547→        vine.lineTo(x + Math.sin(s + i) * 6, vy);
   548→      }
   549→      vine.strokePath();
   550→      // Leaves (small green dots at end)
   551→      vine.fillStyle(0x40a040, 0.6);
   552→      for (let l = 0; l < 3; l++) {
   553→        vine.fillCircle(x + Math.sin(l) * 5, vy + l * 3, 2);
   554→      }
   555→      container.add(vine);
   556→      this.tweens.push(this.scene.tweens.add({
   557→        targets: vine, x: x + 4, duration: 4000 + i * 300, yoyo: true, repeat: -1, ease: 'Sine.inOut',
   558→      }));
   559→    }
   560→
   561→    // Glowing spores (floating, bioluminescent)
   562→    const sporeCount = 30;
   563→    for (let i = 0; i < sporeCount; i++) {
   564→      const sx = Math.random() * this.worldWidth;
   565→      const sy = 50 + Math.random() * (GAME.HEIGHT - 200);

   567→      spore.setBlendMode(Phaser.BlendModes.ADD);
   568→      container.add(spore);
   569→      // Drift upward slowly
   570→      this.tweens.push(this.scene.tweens.add({
   571→        targets: spore, y: sy - 80, x: sx + (Math.random() - 0.5) * 40,
   572→        alpha: { from: 0.6, to: 0 }, duration: 6000 + Math.random() * 4000, repeat: -1,
   573→        delay: Math.random() * 3000, onComplete: (_t, targets) => {
   574→          (targets[0] as Phaser.GameObjects.Arc).setPosition(sx, sy).setAlpha(0.6);
   575→        },
   576→      }));
   577→    }
   578→
   579→    this.layers.push(container);

   588→
   589→    // Foreground ferns (silhouettes at bottom)

   598→      for (let f = 0; f < 5; f++) {
   599→        const ang = -Math.PI / 2 + (f - 2) * 0.4;
   600→        const len = 40 + f * 5;

   603→        fern.lineTo(x + Math.cos(ang) * len, y + Math.sin(ang) * len);

   607→        for (let l = 0; l < 4; l++) {
   608→          const lx = x + Math.cos(ang) * (l * 10 + 5);
   609→          const ly = y + Math.sin(ang) * (l * 10 + 5);
   610→          fern.beginPath();
   611→          fern.moveTo(lx, ly);
   612→          fern.lineTo(lx + Math.cos(ang + Math.PI / 2) * 4, ly + Math.sin(ang + Math.PI / 2) * 4);
   613→          fern.strokePath();

   615→          fern.moveTo(lx, ly);
   616→          fern.lineTo(lx + Math.cos(ang - Math.PI / 2) * 4, ly + Math.sin(ang - Math.PI / 2) * 4);
   617→          fern.strokePath();
   618→        }
   619→        fern.lineStyle(3, 0x1a3a20, 0.8);
   620→      }

   634→      mush.fillStyle(0x80ff80, 0.4); mush.fillEllipse(x, y - 13, 10, 4);
